# KB Subscriptions: DynamoDB tables (Terraform)
# Matches docs/data-model.md. Apply with: terraform init && terraform apply
#
# Tables
#   kb-subscriptions    single table: config, apps, members, subscriptions, match index,
#                       digest buffer, deliveries
#   kb-article-staging  OPTIONAL dev stand-in for your existing staging table
#                       (create_staging_table = true). In prod the real staging table
#                       already exists; only its stream is consumed.

terraform {
  required_version = ">= 1.5"
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = ">= 5.0"
    }
  }
}

provider "aws" {
  region = var.region
}

# ---------- Variables ----------

variable "region" {
  type    = string
  default = "us-east-1"
}

variable "env" {
  type    = string
  default = "dev"
}

variable "table_name" {
  type    = string
  default = "kb-subscriptions"
}

variable "staging_table_name" {
  type    = string
  default = "kb-article-staging"
}

variable "create_staging_table" {
  description = "Create a stand-in staging table (dev/test only)."
  type        = bool
  default     = false
}

variable "deletion_protection" {
  type    = bool
  default = true
}

variable "kms_key_arn" {
  description = "Customer-managed KMS key ARN. Null = AWS-owned key."
  type        = string
  default     = null
}

locals {
  tags = {
    app = "kb-subscriptions"
    env = var.env
  }
}

# ---------- Main table ----------

resource "aws_dynamodb_table" "kb_subscriptions" {
  name                        = var.table_name
  billing_mode                = "PAY_PER_REQUEST"
  hash_key                    = "PK"
  range_key                   = "SK"
  deletion_protection_enabled = var.deletion_protection

  # Stream feeds the reconciler (large match-key diffs) and audit
  stream_enabled   = true
  stream_view_type = "NEW_AND_OLD_IMAGES"

  attribute {
    name = "PK"
    type = "S"
  }
  attribute {
    name = "SK"
    type = "S"
  }
  attribute {
    name = "GSI1PK"
    type = "S"
  }
  attribute {
    name = "GSI1SK"
    type = "S"
  }
  attribute {
    name = "GSI2PK"
    type = "S"
  }
  attribute {
    name = "GSI2SK"
    type = "S"
  }
  attribute {
    name = "GSI3PK"
    type = "S"
  }
  attribute {
    name = "GSI3SK"
    type = "S"
  }

  # GSI1 by user: login (role + apps) and a user's own subscriptions
  global_secondary_index {
    name               = "GSI1"
    hash_key           = "GSI1PK"
    range_key          = "GSI1SK"
    projection_type    = "INCLUDE"
    non_key_attributes = ["name", "role", "appName", "status", "appId", "type", "sourceId", "cadenceId"]
  }

  # GSI2 by source: source counts and impact analysis on filter/priority changes
  global_secondary_index {
    name               = "GSI2"
    hash_key           = "GSI2PK"
    range_key          = "GSI2SK"
    projection_type    = "INCLUDE"
    non_key_attributes = ["name", "status", "appId", "filters", "anchorFilterType", "matchKeys", "version"]
  }

  # GSI3 deliveries by subscription, newest first
  global_secondary_index {
    name               = "GSI3"
    hash_key           = "GSI3PK"
    range_key          = "GSI3SK"
    projection_type    = "INCLUDE"
    non_key_attributes = ["status", "channelId", "articleId", "articleIds", "attempts", "error", "sentAt"]
  }

  # Digest buffers and deliveries expire via expiresAt (epoch seconds)
  ttl {
    attribute_name = "expiresAt"
    enabled        = true
  }

  point_in_time_recovery {
    enabled = true
  }

  server_side_encryption {
    enabled     = true
    kms_key_arn = var.kms_key_arn
  }

  tags = local.tags
}

# ---------- Staging stand-in (dev only) ----------
# TODO(handoff): replace the key schema with the real staging table's once confirmed.

resource "aws_dynamodb_table" "staging" {
  count        = var.create_staging_table ? 1 : 0
  name         = var.staging_table_name
  billing_mode = "PAY_PER_REQUEST"
  hash_key     = "sourceId"
  range_key    = "articleId"

  stream_enabled   = true
  stream_view_type = "NEW_AND_OLD_IMAGES"

  attribute {
    name = "sourceId"
    type = "S"
  }
  attribute {
    name = "articleId"
    type = "S"
  }

  tags = merge(local.tags, { purpose = "staging-stand-in" })
}

# ---------- Outputs ----------

output "table_name" {
  value = aws_dynamodb_table.kb_subscriptions.name
}

output "table_arn" {
  value = aws_dynamodb_table.kb_subscriptions.arn
}

output "table_stream_arn" {
  description = "Reconciler Lambda event source."
  value       = aws_dynamodb_table.kb_subscriptions.stream_arn
}

output "staging_stream_arn" {
  description = "Stream handler Lambda event source (stand-in only)."
  value       = var.create_staging_table ? aws_dynamodb_table.staging[0].stream_arn : null
}
